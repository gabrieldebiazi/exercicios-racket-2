;Exercício 1: ler um número inteiro e dizer se é negativo, zero ou positivo
(define (sinal)
  (display "Digite um número: ")
  (define num (read))
  (cond
    [(> num 0)
         "Positivo"]
    [(< num 0)
         "Negativo"]
    [else "Zero"]
    ))

;(sinal)

;Exercício 2: ler um número, se for maior que 20 retornar sua metade, se não retornar ele próprio
(define (maior20)
  (display "Digite um número: ")
  (define num(read))
  (if (> num 20) (/ num 2) num))

;(maior20)

;Exercício 5: Função de conversão que recebe uma temperatura em celsius e uma string indicando qual notação ela deve ser convertida

(define (conversaoTemp)
  (display "Digite a temperatura atual em ºC: ")
  (define temp (read))
  (read-line)
  (display "Digite a notação para qual você deseja converter: ")
  (define notacao (read-line))
  (cond
    [(equal? notacao "Kelvin")
     (displayln (+ temp 273.15))]
    [(equal? notacao "Fahrenheit")
     (displayln (+ (* temp (/ 9 5)) 32))]
    [else "Notação não reconhecida."]))

;(conversaoTemp)

;Exercício 9: Função que recebe três medidas e verifica se é possível criar um triangulo. caso possivel, verificar se é equilatero, isosceles ou escaleno

(define (triangulo x y z)
  (define T #t)
  (cond
    [(> x (+ y z))
     (set! T #f)]
    [(> y (+ x z))
     (set! T #f)]
    [(> z (+ x y))
     (set! T #f)])

    (if (equal? T #t)
        (cond
          [(and (= x y) (= y z))
           "O triangulo é equilátero"]
          [(and (not (= x y)) (not (= y z)))
           "O triangulo é escaleno"]
          [else "O triangulo é isósceles"])
        "Não é possível formar um triangulo com essas medidas."))

;(triangulo 10 3 2)

;Exercício 12: Função que recebe um ponto no plano cartesiano e retorna em qual quadrante ele se encontra
(define (quadrante x y)
  (cond
    [(and (> x 0) (> y 0))
     "O ponto se encontra no primeiro quadrante."]
    [(and (< x 0) (> y 0))
     "O ponto se encontra no segundo quadrante."]
    [(and (< x 0) (< y 0))
     "O ponto se encontra no terceiro quadrante."]
    [(and (> x 0) (< y 0))
     "O ponto se encontra no quarto quadrante."]
    [else "O ponto se encontra na origem."]))

;Exercício 14a: recebe 3 números e retorna o resultado do maior numero - menor numero
(define (encontrar-menor a b c)
  (min a b c))

(define (encontrar-maior a b c)
  (max a b c))

(define (encontrar-meio a b c)
  (- (+ a b c) 
     (encontrar-menor a b c) 
     (encontrar-maior a b c)))

(define (14a a b c)
  (- (encontrar-maior a b c) (encontrar-menor a b c)))

;(14a 3 1 5)

;Exercício 14b: recebe 3 números e retorna o resultado do numero do meio * menor numero
(define (14b a b c)
  (* (encontrar-meio a b c) (encontrar-menor a b c)))

;(14b 2 5 7)

;Exercício 14a: recebe 3 números e retorna o resultado do numero do meio + maior numero
(define (14c a b c)
  (+ (encontrar-meio a b c) (encontrar-maior a b c)))

;(14c 3 5 7)
